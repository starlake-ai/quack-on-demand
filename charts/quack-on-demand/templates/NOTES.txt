# Port-forward the admin UI / REST

  kubectl --namespace {{ .Release.Namespace }} port-forward \
    svc/{{ include "qod.fullname" . }} {{ .Values.service.rest.port }}:{{ .Values.service.rest.port }}

  Then open http://localhost:{{ .Values.service.rest.port }}/ui/

  Seeded admin user(s): {{ .Values.admin.username }}
{{- if .Values.admin.existingSecret }}
  Password: (from {{ .Values.admin.existingSecret }}/{{ .Values.admin.existingSecretKey }})
{{- else }}
  Password: (chart-managed; from values.admin.password)
{{- end }}
  (used only when the admin row does not exist yet)

  From a shell, with the qod CLI (pip install qod):

    qod login --url http://localhost:{{ .Values.service.rest.port }} --username {{ splitList "," .Values.admin.username | first | trim }}

  This chart sets no static API key: automation (CI, MCP clients, SCIM
  connectors) uses a personal access token, minted once from a session:

    qod auth pat create --name ci
{{- $security := .Values.security | default dict }}
{{- if and (not (dig "nodeLockdown" true $security)) (not (dig "aclEnabled" false $security)) }}

  WARNING: security.nodeLockdown and security.aclEnabled are both off. Any
  tenant user can read the node's mounted secrets, including the Postgres
  password of postgres.user ({{ .Values.postgres.user }}), and with it the
  control-plane database and every tenant's catalog. See "Tenant isolation"
  in the chart README.
{{- end }}

# Port-forward the native Quack protocol front door (DuckDB clients)

  kubectl --namespace {{ .Release.Namespace }} port-forward \
    svc/{{ include "qod.fullname" . }}-quack {{ .Values.service.quack.port }}:{{ .Values.service.quack.port }}

  From any DuckDB with the quack extension (replace <user>/<password>/<tenant>/<pool>):

    ATTACH 'quack:localhost:{{ .Values.service.quack.port }}' AS qod (TYPE quack, TOKEN 'tenant=<tenant>&pool=<pool>&user=<user>&password=<password>');
{{- end }}

# Notes

- Replicas default to 1 (Recreate rollout). Set replicaCount > 1 to enable
  active-active HA (QOD_HA_ENABLED); requires sessionJwtSecret or
  existingSessionJwtSecret. See guides/RESILIENCE.md.
- The manager calls the K8s API to create / delete pods + services in
  this namespace. RBAC scope is a Role, not a ClusterRole.
- {{- if eq .Values.metrics.sink "prometheus" }} /metrics is exposed on the rest-ui port. {{- else }} metrics sink: {{ .Values.metrics.sink }} (the /metrics endpoint is NOT exposed in cloud-push mode). {{- end }}
